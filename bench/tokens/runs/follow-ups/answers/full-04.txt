state status="All"
derive shown={status == 'All' ? claims : filter(claims, 'status', status)}
derive totalIncurred={sum(claims, 'incurred')}
derive openCount={count(claims, 'status', 'Open')}
derive reviewCount={count(claims, 'status', 'In review')}
derive avgDays={avg(claims, 'days')}
text "Claims review, 2025" style=title
grid min=170 gap=12
  card pad=18 gap=10
    stat label="Total claims" value={len(claims)}
  card pad=18 gap=10
    stat label="Total incurred" value={compact(totalIncurred, 'USD')}
  card pad=18 gap=10
    stat label="Open claims" value={openCount} tone=accent change={reviewCount + ' in review'}
  card pad=18 gap=10
    stat label="Avg days open" value={round(avgDays)}
grid min=300 gap=12
  card pad=18 gap=10
    chart "Incurred by month" kind=bar items={monthly} x=month y=incurred format=money
  card pad=18 gap=10
    chart "Claims opened and closed by month" kind=line items={monthly} x=month y=opened,closed
  card pad=18 gap=10
    chart "Incurred by line" kind=donut items={byLine} x=line y=incurred format=money
card pad=18 gap=10
  text "All claims" style=title
  segmented bind=status label="Filter by status" options="All,Open,In review,Closed,Denied"
  text "{len(shown)} of {len(claims)} claims" tone=muted
  table "Claims" items={shown} key=id
    table-column "ID" field=id
    table-column "Opened" field=opened
    table-column "Line" field=line
    table-column "Site" field=site
    table-column "Status" field=status
    table-column "Incurred" field=incurred format=money align=right
    table-column "Days" field=days format=number align=right
